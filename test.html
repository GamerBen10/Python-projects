# Define a simple function to check for prime numbers
def is_prime(n: int) -> bool:
    if n <= 1:
        return False
    for i in range(2, int(n**0.5) + 1):
        if n % i == 0:
            return False
    return True

# Generate primes up to 20
primes = [num for num in range(20) if is_prime(num)]
print(f"Prime numbers under 20: {primes}")
